#!/usr/bin/env python3
"""No-input real-agent viewport resize and explicit fux termination evidence."""
import argparse
from datetime import datetime, timezone
import json
from pathlib import Path
import subprocess
import tempfile
import time

from capture_startup import rpc
from capture_opencode_events import sha256


def validate(value):
    assert value['server_exit'] == 0 and value['input_sent'] is False
    stages = value['stages']
    assert [s['name'] for s in stages] == ['initial', 'narrow', 'restored']
    first, narrow, restored = [s['captures'][-1] for s in stages]
    assert first['columns'] > narrow['columns'] and first['rows'] == narrow['rows']
    assert (restored['rows'], restored['columns']) == (first['rows'], first['columns'])
    assert any(c['text'] for c in stages[0]['captures'])
    for stage in stages:
        assert len(stage['captures']) <= 32
        assert stage['instance'] == value['instance'] and stage['pane'] == value['pane']['id']
        for capture in stage['captures']:
            assert capture['input_sequence'] == 0
    final = value['final']
    assert final['pane'] == value['pane']['id'] and final['input_sequence'] == 0
    assert final['command'] == [value['agent']]
    assert value['exit_trigger'] == 'explicit fux kill; not natural completion'


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    for name in ['fux', 'agent', 'output']:
        parser.add_argument('--'+name, type=Path, required=True)
    args = parser.parse_args()
    fux, agent = args.fux.resolve(strict=True), args.agent.absolute()
    agent.resolve(strict=True)
    args.output.mkdir(mode=0o700, parents=True, exist_ok=False)
    value = dict(schema=1, captured_at=datetime.now(timezone.utc).isoformat(), agent=str(agent),
        binary_sha256=sha256(agent), fux_sha256=sha256(fux), harness_sha256=sha256(Path(__file__)),
        input_sent=False, stages=[], exit_trigger='explicit fux kill; not natural completion',
        limitation='Fresh unauthenticated HOME/XDG; viewport snapshots, not working or task-completion evidence.')
    with tempfile.TemporaryDirectory(prefix='zlife-',dir='/tmp') as directory:
        root = Path(directory)
        for name in ['config/fux','state','cache','data','codex','claude','work']:
            (root/name).mkdir(parents=True)
        env = dict(PATH='/usr/bin:/bin:/usr/sbin:/sbin:/opt/homebrew/bin', HOME=directory,
            XDG_RUNTIME_DIR=directory, XDG_CONFIG_HOME=str(root/'config'), XDG_STATE_HOME=str(root/'state'),
            XDG_DATA_HOME=str(root/'data'), XDG_CACHE_HOME=str(root/'cache'), CODEX_HOME=str(root/'codex'),
            CLAUDE_CONFIG_DIR=str(root/'claude'), SHELL='/bin/sh', TERM='xterm-256color', LANG='en_US.UTF-8')
        value['version'] = subprocess.run([str(agent),'--version'],env=env,cwd=root/'work',
            capture_output=True,check=True,timeout=10).stdout.decode().strip()
        (root/'config/fux/config.toml').write_text('default-command = { argv = '+json.dumps([str(agent)])+' }\n')
        process = subprocess.Popen([str(fux),'serve'],env=env,cwd=root/'work',stdin=subprocess.DEVNULL,
            stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL)
        control = root/'fux/default.sock'
        def until(check):
            deadline = time.monotonic()+20
            while time.monotonic()<deadline:
                assert process.poll() is None, 'owned fux exited'
                result = check()
                if result: return result
                time.sleep(.1)
            raise TimeoutError('lifecycle observation deadline')
        try:
            until(control.exists)
            listing = rpc(control,dict(id=1,command='list'))
            value['instance'] = listing['instance']
            value['pane'] = listing['workspaces'][0]['tabs'][0]['panes'][0]
            pane = value['pane']['id']
            def command(name, **fields):
                return rpc(control,dict(id=1,command=name,instance=value['instance'],**fields))
            def capture(): return command('capture',pane=pane,max_bytes=131072)
            until(lambda: capture()['text'])
            for name in ['initial','narrow','restored']:
                if name == 'narrow':
                    other = command('split',axis='horizontal',target=pane,argv=['/bin/cat'])['pane']
                elif name == 'restored': command('kill',pane=other)
                captures = []
                for _ in range(15):
                    captures.append(capture()); time.sleep(.1)
                value['stages'].append(dict(name=name,instance=value['instance'],pane=pane,captures=captures))
            command('kill',pane=pane)
            def final():
                result = subprocess.run([str(fux),'final','--instance',value['instance'],str(pane)],
                    env=env,capture_output=True,timeout=3)
                if result.returncode:
                    value['final_problem'] = (result.stdout+result.stderr).decode(errors='replace')[:1024]
                    return None
                return json.loads(result.stdout)['result']['value']['record']
            value['final'] = until(final)
        finally:
            try:
                if process.poll() is None: process.terminate()
                try: value['server_exit'] = process.wait(timeout=10)
                except subprocess.TimeoutExpired:
                    process.kill(); process.wait(timeout=5); raise
            finally:
                (args.output/'diagnostic.json').write_text(json.dumps(value,indent=2).replace(directory,'<CAPTURE_ROOT>')+'\n')
        validate(value)
        (args.output/'lifecycle.json').write_text(json.dumps(value,indent=2).replace(directory,'<CAPTURE_ROOT>')+'\n')
    print(args.output/'lifecycle.json')


if __name__ == '__main__': main()
