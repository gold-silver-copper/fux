#!/usr/bin/env python3
"""Bounded real Codex unanswered-approval capture using an authorized login cache.

Uses a fresh HOME/config/work directory, copies only auth.json, and requests approval for /usr/bin/true.
The harness never answers the command approval. Never run in unattended CI: this consumes the selected account's usage.
The output is viewport evidence, not proof of task completion or native binding.
"""
import argparse
from datetime import datetime, timezone
import json
import os
from pathlib import Path
import re
import subprocess
import tempfile
import time

from capture_startup import rpc
from capture_opencode_events import sha256

PROMPT = 'This is an approval UI test. Request approval to run /usr/bin/true outside the sandbox, with justification: May I run the harmless approval fixture? Do not execute any other command or tool. Wait for the user to approve; do not bypass approval.'


def strings(value):
    if isinstance(value, str):
        yield value
    elif isinstance(value, dict):
        for item in value.values():
            yield from strings(item)
    elif isinstance(value, list):
        for item in value:
            yield from strings(item)


def validate(value):
    assert value['provider'] == 'real authenticated OpenAI; no synthetic provider'
    assert value['server_exit'] == 0 and not value.get('forced_server_cleanup')
    assert not value.get('failure')
    assert value['exit_trigger'] == 'explicit fux kill; approval unanswered'
    assert [item['keys'] for item in value['inputs']] == ['\r']
    final = value['final']
    assert final['pane'] == value['pane'] and final['command'] == value['argv']
    assert final['exit_status'] is None and final['input_sequence'] == 1
    captures = value['captures']
    assert 0 < len(captures) <= 600
    assert all(item['capture']['rows'] == 23 and item['capture']['columns'] == 80 for item in captures)
    assert any('Do you trust the contents of this directory?' in item['capture']['text'] for item in captures)
    last = captures[-1]['capture']
    assert last['input_sequence'] == 1
    assert 'Would you like to run the following command?' in last['text']
    assert 'Press enter to confirm or esc to cancel' in last['text']
    assert 'Reason: May I run the harmless approval fixture?' in last['text']
    assert '$ /usr/bin/true' in last['text']
    assert '3. No, and tell Codex what to do differently (esc)' in last['text']


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    for name in ['fux', 'agent', 'auth-file', 'output']:
        parser.add_argument('--'+name, type=Path, required=True)
    parser.add_argument('--model', required=True)
    args = parser.parse_args()
    fux, agent = args.fux.resolve(strict=True), args.agent.absolute()
    agent.resolve(strict=True)
    auth = args.auth_file.read_bytes()
    sensitive = [s for s in strings(json.loads(auth)) if len(s) >= 12]
    args.output.mkdir(mode=0o700, parents=True, exist_ok=False)
    evidence = dict(schema=1, captured_at=datetime.now(timezone.utc).isoformat(),
        agent_version='', binary_sha256=sha256(agent), fux_sha256=sha256(fux),
        harness_sha256=sha256(Path(__file__)), model=args.model, prompt=PROMPT,
        provider='real authenticated OpenAI; no synthetic provider',
        isolation='Fresh HOME/XDG/CODEX_HOME/work; only authorized auth.json copied; read-only sandbox, approval on-request; approval is not answered.',
        limitation='Passive viewport observations; no native prompt binding or verified task outcome.',
        captures=[], inputs=[], exit_trigger='explicit fux kill; approval unanswered', final=None)
    with tempfile.TemporaryDirectory(prefix='zca-', dir='/tmp') as directory:
        root = Path(directory)
        for name in ['config/fux', 'codex', 'work', 'state', 'cache', 'data']:
            (root/name).mkdir(parents=True)
        auth_path = root/'codex/auth.json'
        with auth_path.open('xb') as handle:
            os.chmod(auth_path, 0o600)
            handle.write(auth)
        (root/'codex/config.toml').write_text(
            'cli_auth_credentials_store = "file"\ncheck_for_update_on_startup = false\n')
        env = dict(PATH='/usr/bin:/bin:/usr/sbin:/sbin:/opt/homebrew/bin', HOME=directory,
            XDG_RUNTIME_DIR=directory, XDG_CONFIG_HOME=str(root/'config'),
            XDG_STATE_HOME=str(root/'state'), XDG_DATA_HOME=str(root/'data'),
            XDG_CACHE_HOME=str(root/'cache'), CODEX_HOME=str(root/'codex'),
            SHELL='/bin/sh', TERM='xterm-256color', LANG='en_US.UTF-8')
        argv = [str(agent), '-m', args.model, '-s', 'read-only', '-a', 'on-request', '--no-alt-screen', PROMPT]
        evidence['argv'] = argv
        evidence['agent_version'] = subprocess.run([str(agent), '--version'], env=env,
            cwd=root/'work', capture_output=True, check=True, timeout=10).stdout.decode().strip()
        (root/'config/fux/config.toml').write_text('default-command = { argv = '+json.dumps(argv)+' }\n')
        process = subprocess.Popen([str(fux), 'serve'], env=env, cwd=root/'work',
            stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        control = root/'fux/default.sock'
        pane = instance = None
        started = time.monotonic()
        failed = None
        try:
            while not control.exists():
                if process.poll() is not None or time.monotonic()-started > 10:
                    raise RuntimeError('fux startup failed')
                time.sleep(.1)
            listing = rpc(control, dict(id=1, command='list'))
            instance = listing['instance']
            pane = listing['workspaces'][0]['tabs'][0]['panes'][0]['id']
            evidence.update(instance=instance, pane=pane)
            def command(name, **fields):
                return rpc(control, dict(id=1, command=name, instance=instance, **fields))
            def submit(keys, purpose):
                receipt = command('input-reserve', pane=pane)['receipt']
                result = command('input-submit', operation=receipt['operation'], keys=keys)
                evidence['inputs'].append(dict(purpose=purpose, keys=keys, result=result))
            response_at = None
            while time.monotonic()-started < 60:
                capture = command('capture', pane=pane, max_bytes=131072)
                text = capture['text']
                if not evidence['captures'] or text != evidence['captures'][-1]['capture']['text']:
                    evidence['captures'].append(dict(elapsed_ms=int((time.monotonic()-started)*1000), capture=capture))
                if ('Do you trust the contents of this directory?' in text
                        and '1. Yes, continue' in text and not evidence['inputs']):
                    time.sleep(.5)
                    submit('\r', 'trust only the disposable fixture directory')
                if 'Would you like to run the following command?' in text or 'Press enter to confirm or esc to cancel' in text or 'Allow command?' in text:
                    if response_at is None:
                        response_at = time.monotonic()
                    if time.monotonic()-response_at > 2:
                        break
                time.sleep(.1)
            else:
                raise TimeoutError('no observed approval within 60 seconds')
            command('kill', pane=pane)
            deadline = time.monotonic()+10
            while time.monotonic() < deadline:
                result = subprocess.run([str(fux), 'final', '--instance', instance, str(pane)],
                    env=env, cwd=root/'work', capture_output=True, timeout=3)
                if result.returncode == 0:
                    evidence['final'] = json.loads(result.stdout)['result']['value']['record']
                    evidence.pop('final_problem', None)
                    break
                try:
                    evidence['after_quit'] = command('capture', pane=pane, max_bytes=131072)
                except (OSError, ValueError, EOFError):
                    pass
                evidence['final_problem'] = (result.stdout+result.stderr).decode(errors='replace')[:1024]
                time.sleep(.1)
            if evidence['final'] is None:
                raise TimeoutError('natural exit final record missing')
        except Exception as error:
            failed = type(error).__name__
            evidence['failure'] = failed
        finally:
            if evidence['final'] is None and pane is not None and process.poll() is None:
                try:
                    rpc(control, dict(id=1, command='kill', instance=instance, pane=pane))
                except (OSError, ValueError, EOFError):
                    pass
            if process.poll() is None:
                process.terminate()
            try:
                process.wait(timeout=10)
            except subprocess.TimeoutExpired:
                process.kill()
                process.wait(timeout=5)
                evidence['forced_server_cleanup'] = True
                failed = 'CleanupTimeout'
            evidence['server_exit'] = process.returncode
            if process.returncode != 0:
                failed = 'ServerExit'
            # Include refreshed credentials in the leak check without publishing them.
            if auth_path.exists():
                sensitive.extend(s for s in strings(json.loads(auth_path.read_bytes())) if len(s) >= 12)
            encoded = json.dumps(evidence, indent=2, ensure_ascii=False)
            if any(secret in encoded for secret in sensitive):
                raise RuntimeError('credential material detected; refusing publication')
            encoded = encoded.replace(str(root.resolve()), '<CAPTURE_ROOT>').replace(directory, '<CAPTURE_ROOT>')
            encoded = re.sub(r'[\w.+-]+@[\w.-]+', '<EMAIL>', encoded)
            if not failed:
                try:
                    validate(evidence)
                except (AssertionError, KeyError):
                    failed = 'EvidenceValidation'
            destination = args.output/('diagnostic.json' if failed else 'approval.json')
            destination.write_text(encoded+'\n')
        if failed:
            raise RuntimeError(f'{failed}; sanitized diagnostic retained')
    print(destination)


if __name__ == '__main__':
    main()
