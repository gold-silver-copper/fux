#!/usr/bin/env python3
"""Headless fux/zor viewer baseline. Synthetic workload; no accounts or network endpoints."""
import argparse
import hashlib
import json
import os
from pathlib import Path
import platform
import select
import socket
import struct
import subprocess
import tempfile
import threading
import time

from comparisons import prompt_boundary as common

ROOT = Path(__file__).resolve().parents[1]
WORKER = r'''
#include <stdio.h>
#include <string.h>
#include <unistd.h>
int main(void) {
    alarm(90); setbuf(stdout, NULL); char line[64]; puts("READY");
    while (fgets(line, sizeof line, stdin)) {
        line[strcspn(line, "\r\n")] = 0;
        int slow = line[0] == 'S';
        for (int i = 0; i < (slow ? 100 : 1000); i++) {
            printf("%04d abcdefghijklmnopqrstuvwxyz\n", i);
            if (slow) usleep(10000);
        }
        printf("DONE_%s\n", line);
    }
    return 0;
}
'''


def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


class Viewer:
    def __init__(self, path, slow):
        self.peer = socket.socket(socket.AF_UNIX)
        self.peer.settimeout(3)
        self.peer.connect(str(path))
        data = json.dumps(dict(type='hello', version=6, rows=24, columns=80)).encode()
        self.peer.sendall(struct.pack('>I', len(data)) + data)
        self.peer.setblocking(False)
        self.slow = slow
        self.stop = threading.Event()
        self.lock = threading.Lock()
        self.received_bytes = self.frame_bytes = self.frames = self.pending_high_water = 0
        self.latest = {}; self.errors = []
        self.thread = threading.Thread(target=self.read)
        self.thread.start()

    def read(self):
        pending = bytearray()
        try:
            while not self.stop.is_set():
                if not select.select([self.peer], [], [], .1)[0]:
                    continue
                chunk = self.peer.recv(65536)
                if not chunk:
                    raise EOFError('viewer disconnected')
                pending.extend(chunk)
                with self.lock:
                    self.received_bytes += len(chunk)
                    self.pending_high_water = max(self.pending_high_water, len(pending))
                while len(pending) >= 4:
                    length = struct.unpack('>I', pending[:4])[0]
                    assert 0 < length <= 16 * 1024 * 1024
                    if len(pending) < length + 4:
                        break
                    value = json.loads(pending[4:4 + length]); del pending[:4 + length]
                    with self.lock:
                        self.frame_bytes += length + 4
                        if 'state' in value:
                            self.frames += 1
                            self.latest = value['state']['state']
                        elif value != {'hello': {'version': 6}}:
                            raise AssertionError(value)
                    if self.slow:
                        self.stop.wait(.025)
        except BaseException as error:
            if not self.stop.is_set():
                self.errors.append(repr(error))

    def counters(self):
        with self.lock:
            return dict(received_bytes=self.received_bytes, decoded_frame_bytes=self.frame_bytes,
                        frames=self.frames, pending_high_water_bytes=self.pending_high_water)

    def visible(self, marker, count):
        with self.lock:
            panes = self.latest.get('panes', {})
            return len(panes) == count and all(marker in ''.join(c['text'] for c in p['cells']) for p in panes.values())

    def close(self):
        self.stop.set(); self.thread.join(timeout=3)
        self.peer.close()
        assert not self.thread.is_alive(), 'viewer reader did not stop'
        assert not self.errors, self.errors


def run_case(fux, zor, worker, sampler, panes, viewers, slow):
    with tempfile.TemporaryDirectory(prefix='hperf-', dir='/tmp') as directory:
        root = Path(directory); config = root/'config/fux'; config.mkdir(parents=True)
        (config/'config.toml').write_text('default-command = { argv = ' + json.dumps([str(worker)]) + ' }\n')
        env = dict(PATH='/usr/bin:/bin', HOME=directory, XDG_RUNTIME_DIR=directory,
                   XDG_CONFIG_HOME=str(root/'config'), XDG_STATE_HOME=str(root/'state'), TERM='xterm-256color')
        owners = {}; readers = []; worker_pids = []; row = dict(panes=panes, viewers=viewers, slow=slow, geometry=[80,24], phases=[])
        def until(check):
            deadline = time.monotonic() + 12
            while time.monotonic() < deadline:
                assert all(p.poll() is None for p in owners.values()), 'owner exited'
                assert all(not r.errors for r in readers), [r.errors for r in readers]
                if check(): return
                time.sleep(.01)
            raise TimeoutError('performance fixture observation')
        def sample():
            values = {}
            for name, process in owners.items():
                if sampler:
                    values[name] = json.loads(subprocess.run([str(sampler), str(process.pid)], capture_output=True, check=True, timeout=3).stdout)
                else:
                    values[name] = None
            return dict(monotonic=time.monotonic(), owners=values, viewers=[r.counters() for r in readers])
        try:
            owners['fux'] = subprocess.Popen([str(fux), 'serve'], cwd=root, env=env, stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
            control = root/'fux/default.sock'
            until(control.exists)
            for _ in range(panes - 1):
                common.rpc(control, dict(command='split', id=1, axis='horizontal', argv=[str(worker)]), True)
            for index in range(viewers):
                readers.append(Viewer(root/'fux/default.attach.sock', slow and index == viewers - 1))
            until(lambda: all(r.visible('READY', panes) for r in readers))
            listing = common.rpc(control, dict(command='list', id=1), True)['result']['value']
            pane_ids = [p['id'] for w in listing['workspaces'] for t in w['tabs'] for p in t['panes']]
            worker_pids = [p['pid'] for w in listing['workspaces'] for t in w['tabs'] for p in t['panes']]
            assert all(isinstance(pid, int) and pid > 1 for pid in worker_pids)
            row['pane_rects'] = [p['geometry'] for w in listing['workspaces'] for t in w['tabs'] for p in t['panes']]
            owners['zor'] = subprocess.Popen([str(zor), 'serve'], cwd=root, env=env, stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
            until(lambda: (root/'zor/control.sock').exists())
            def observed():
                reply = subprocess.run([str(zor), 'status'], cwd=root, env=env, capture_output=True, timeout=3)
                if reply.returncode: return False
                value = json.loads(reply.stdout)
                return len(value.get('snapshot', {}).get('observations', [])) == panes and not value.get('stale', True)
            until(observed)
            time.sleep(.3)
            for label in ['idle', 'B01', 'S01']:
                before = sample(); start = time.monotonic()
                if label == 'idle':
                    time.sleep(1)
                else:
                    for pane in pane_ids:
                        common.rpc(control, dict(command='send-keys', id=1, pane=pane, keys=label+'\\n'), True)
                    until(lambda: all(r.visible('DONE_'+label, panes) for r in readers))
                visible_ms = (time.monotonic() - start) * 1000
                time.sleep(.1)
                row['phases'].append(dict(name=label, visible_ms=None if label == 'idle' else visible_ms,
                                         workload_ms=visible_ms, before=before, after=sample()))
            row['passed'] = True
        finally:
            failures = []
            for reader in readers:
                try: reader.close()
                except BaseException as error: failures.append(repr(error))
            exits = {}
            for name, process in reversed(list(owners.items())):
                try:
                    process.terminate(); exits[name] = process.wait(timeout=10)
                    if exits[name] != 0: failures.append((name, exits[name]))
                except BaseException as error:
                    failures.append(repr(error))
                    try: process.kill(); process.wait(timeout=3)
                    except BaseException as fallback: failures.append(repr(fallback))
            row['owner_exits'] = exits
            deadline = time.monotonic() + 5
            remaining = set(worker_pids)
            while remaining and time.monotonic() < deadline:
                for pid in tuple(remaining):
                    try: os.kill(pid, 0)
                    except ProcessLookupError: remaining.remove(pid)
                if remaining: time.sleep(.02)
            row['worker_exit_confirmed'] = not remaining
            if remaining: failures.append(('workers remain', sorted(remaining)))
            assert not failures, failures
        return row


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    for name in ['fux', 'zor', 'output']: parser.add_argument('--'+name, type=Path, required=True)
    parser.add_argument('--repetitions', type=int, default=3)
    args = parser.parse_args()
    assert 1 <= args.repetitions <= 5 and not args.output.exists()
    fux, zor = args.fux.resolve(strict=True), args.zor.resolve(strict=True)
    report = dict(schema=1, geometry=[80,24], platform=platform.platform(), synthetic=True,
                  provenance=dict(harness_sha256=digest(__file__), fux_sha256=digest(fux), zor_sha256=digest(zor)), results=[])
    with tempfile.TemporaryDirectory(prefix='hperf-tools-', dir='/tmp') as directory:
        root = Path(directory); source = root/'worker.c'; source.write_text(WORKER)
        worker = root/'worker'
        subprocess.run(['cc', str(source), '-o', str(worker)], check=True, timeout=30)
        sampler = None
        if platform.system() == 'Darwin':
            sampler = root/'sample'
            sampler_source = ROOT/'tools/comparisons/resource_sampler.c'
            subprocess.run(['cc', str(sampler_source), '-o', str(sampler)], check=True, timeout=30)
            report['calibration'] = json.loads(subprocess.run([str(sampler), '--calibrate'], capture_output=True, check=True, timeout=3).stdout)
            report['provenance']['sampler_sha256'] = digest(sampler_source)
        report['resource_limit'] = 'Owned-process CPU/RSS only; unavailable on this harness platform if null. No allocation or internal queue counters.'
        for repetition in range(args.repetitions):
            for panes, viewers, slow in [(1,1,False), (1,4,False), (4,4,False), (4,4,True)]:
                result = run_case(fux, zor, worker, sampler, panes, viewers, slow)
                result['repetition'] = repetition + 1; report['results'].append(result)
                print(panes, viewers, slow, repetition + 1, 'passed', flush=True)
    args.output.parent.mkdir(parents=True, exist_ok=True)
    args.output.write_text(json.dumps(report, indent=2)+'\n')


if __name__ == '__main__': main()
