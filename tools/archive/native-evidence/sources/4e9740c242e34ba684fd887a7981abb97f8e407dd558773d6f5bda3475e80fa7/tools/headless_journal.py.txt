#!/usr/bin/env python3
"""Account-free public-CLI journal cost sample; not an internal lock-time profiler."""
import argparse
import hashlib
import json
import os
from pathlib import Path
import resource
import subprocess
import tempfile
import time

from comparisons.prompt_boundary import rpc


def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    for name in ['fux', 'zor', 'output']:
        parser.add_argument('--' + name, type=Path, required=True)
    args = parser.parse_args()
    assert not args.output.exists()
    fux, zor = args.fux.resolve(strict=True), args.zor.resolve(strict=True)
    evidence = dict(schema=1, scope='Public CLI, journal activity, fixed binaries; no model calls',
                    provenance={'fux_sha256': digest(fux), 'zor_sha256': digest(zor),
                                'harness_sha256': digest(__file__)}, runs=[])
    for repetition in range(3):
        with tempfile.TemporaryDirectory(prefix='hjournal-', dir='/tmp') as directory:
            root = Path(directory)
            config = root / 'config/fux'; config.mkdir(parents=True)
            (config/'config.toml').write_text('default-command = { argv = ["/bin/cat"] }\n')
            env = dict(PATH='/usr/bin:/bin', HOME=directory, XDG_RUNTIME_DIR=directory,
                       XDG_CONFIG_HOME=str(root/'config'), XDG_STATE_HOME=str(root/'state'), TERM='xterm-256color')
            server = subprocess.Popen([str(fux), 'serve'], cwd=root, env=env, stdin=subprocess.DEVNULL,
                                      stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
            record = dict(repetition=repetition+1, samples=[])
            worker_pid = None
            try:
                control = root/'fux/default.sock'
                deadline = time.monotonic()+10
                while not control.exists():
                    assert server.poll() is None and time.monotonic()<deadline
                    time.sleep(.02)
                listing = rpc(control, {'command':'list','id':1}, True)['result']['value']
                pane = listing['workspaces'][0]['tabs'][0]['panes'][0]
                worker_pid = pane['pid']
                journal = root/'state/zor/journal.json'
                def command(kind, *argv):
                    before = journal.stat() if journal.exists() else None
                    cpu = resource.getrusage(resource.RUSAGE_CHILDREN)
                    start = time.monotonic()
                    output = subprocess.run([str(zor),'task',*argv], env=env, cwd=root,
                                            capture_output=True, check=True, timeout=10)
                    elapsed = (time.monotonic()-start)*1000
                    after_cpu = resource.getrusage(resource.RUSAGE_CHILDREN)
                    value = json.loads(output.stdout)
                    after = journal.stat()
                    changed = before is None or before.st_ino != after.st_ino
                    record['samples'].append(dict(kind=kind, elapsed_ms=elapsed,
                        child_cpu_ms=(after_cpu.ru_utime+after_cpu.ru_stime-cpu.ru_utime-cpu.ru_stime)*1000,
                        journal_bytes=after.st_size, committed_replacement=changed,
                        logical_committed_bytes=after.st_size if changed else 0))
                    return value
                for index in range(32):
                    command('adopt', 'adopt', 't'+str(index), '--title','journal workload',
                            '--instance',listing['instance'],'--workspace','default','--pane',str(pane['id']))
                for _ in range(32):
                    command('inspect', 'inspect','t0')
                for index in range(32):
                    command('idempotent-adopt', 'adopt', 't'+str(index), '--title','journal workload',
                            '--instance',listing['instance'],'--workspace','default','--pane',str(pane['id']))
                assert all(s['committed_replacement'] == (s['kind']=='adopt') for s in record['samples'])
                record['final_generation'] = json.loads(journal.read_bytes())['generation']
                assert record['final_generation'] == 32
            finally:
                server.terminate()
                try:
                    record['server_exit'] = server.wait(timeout=10)
                except subprocess.TimeoutExpired:
                    server.kill(); server.wait(timeout=3)
                    raise
                assert record['server_exit'] == 0
                if worker_pid:
                    deadline = time.monotonic()+5
                    while True:
                        try: os.kill(worker_pid, 0)
                        except ProcessLookupError: break
                        assert time.monotonic()<deadline, 'worker remains'
                        time.sleep(.02)
                record['worker_exit_confirmed'] = worker_pid is not None
            evidence['runs'].append(record)
            print('journal repetition', repetition+1, 'passed', flush=True)
    evidence['limits'] = ('CPU includes CLI startup; logical committed bytes count atomic journal replacements, '
        'not physical filesystem writes. Inode checks observe these serial operations only. No internal lock '
        'hold time, allocation count or durability timing isolation. Inspection/replay make zero replacements. '
        'The journal implementation is unchanged by the fux sharing optimization.')
    args.output.parent.mkdir(parents=True, exist_ok=True)
    args.output.write_text(json.dumps(evidence, indent=2)+'\n')


if __name__ == '__main__':
    main()
