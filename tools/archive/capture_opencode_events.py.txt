#!/usr/bin/env python3
"""Probe real OpenCode TUI hooks through fux with a deterministic local provider.

This is adapter-contract evidence, not model quality or authenticated-cloud coverage.
Uses private configuration and only submits fixed fixture prompts to owned processes.
"""
import argparse
from datetime import datetime, timezone
import hashlib
from http.server import BaseHTTPRequestHandler, HTTPServer
import json
import os
from pathlib import Path
import subprocess
import tempfile
import threading
import time

from capture_startup import rpc


PLUGIN = r'''
import { appendFileSync } from "node:fs";
export const ZorEventProbe = async () => {
  let bytes = 0;
  const log = (value) => {
    const text = JSON.stringify(value) + "\n";
    bytes += Buffer.byteLength(text);
    if (bytes > 1048576) throw new Error("fixture event limit");
    appendFileSync(process.env.ZOR_PROBE_LOG, text, { mode: 0o600 });
  };
  log({ hook: "loaded" });
  return {
    "chat.message": async (input, output) => log({ hook: "chat.message", input, output }),
    event: async ({ event }) => {
      if (["message.updated", "message.part.updated", "session.status", "session.idle",
           "session.error", "permission.asked", "permission.replied"].includes(event.type))
        log({ hook: "event", event });
    },
  };
};
'''

PROMPTS = ["Reply with FIXTURE RESPONSE", "ZOR_TOOL_STOP: inspect the fixture"]


def sha256(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def read_events(path):
    if not path.exists():
        return []
    with path.open('rb') as stream:
        data = stream.read(1048577)
    if len(data) > 1048576:
        raise ValueError('fixture event limit')
    # The producer may be in appendFileSync; ignore its unfinished final line.
    return [json.loads(line) for line in data.split(b'\n')[:-1]]


def validate(events, submissions, requests, fixture_path='<FIXTURE_ROOT>/work/fixture.txt'):
    chats = [(i, row) for i, row in enumerate(events) if row['hook'] == 'chat.message']
    assert len(chats) == 2, 'expected exactly two consumed prompts'
    assert len(submissions) == 2, 'missing fux submissions'
    operations = [s['receipt']['operation'] for s in submissions]
    assert len(set(operations)) == 2, 'input operation reused across prompts'
    ids = [row['output']['message']['id'] for _, row in chats]
    assert len(set(ids)) == 2, 'user message ID reused'
    session = chats[0][1]['output']['message']['sessionID']
    outcomes = []
    for turn, (chat_index, chat) in enumerate(chats):
        receipt = submissions[turn]['final_receipt']
        assert receipt['operation'] == operations[turn] and receipt['state'] == 'delivered'
        assert receipt['input_sequence'] == turn + 1, 'unexpected input intervened'
        assert receipt['bytes_written'] == len((PROMPTS[turn] + '\r').encode()), 'partial delivery'
        assert chat['output']['message']['sessionID'] == session, 'session changed'
        assert [p['text'] for p in chat['output']['parts'] if p['type'] == 'text'] == [PROMPTS[turn]]
        updates = [(i, row['event']['properties']['info']) for i, row in enumerate(events)
                   if row.get('event', {}).get('type') == 'message.updated'
                   and row['event']['properties']['info'].get('role') == 'assistant'
                   and row['event']['properties']['info'].get('parentID') == ids[turn]
                   and row['event']['properties']['info'].get('sessionID') == session]
        completed = [(i, m) for i, m in updates
                     if m.get('time', {}).get('completed') and m.get('finish') == 'stop'
                     and not m.get('error')]
        assert completed and completed[0][0] > chat_index, 'missing correlated response'
        idle = [i for i, row in enumerate(events)
                if row.get('event', {}).get('type') == 'session.idle'
                and row['event']['properties']['sessionID'] == session and i > completed[-1][0]]
        assert idle, 'missing later idle'
        if turn == 0:
            assert idle[0] < chats[1][0], 'second prompt preceded first idle'
        tool_messages = {row['event']['properties']['part']['messageID'] for row in events
                         if row.get('event', {}).get('type') == 'message.part.updated'
                         and row['event']['properties']['part'].get('type') == 'tool'
                         and row['event']['properties']['part'].get('sessionID') == session}
        if turn == 1:
            assert len({m['id'] for _, m in completed}) >= 2, 'tool-stop did not continue'
            assert completed[0][1]['id'] in tool_messages, 'missing intermediate tool message'
            assert completed[-1][1]['id'] not in tool_messages, 'missing final text response'
            assert any(row.get('event', {}).get('type') == 'message.part.updated'
                       and row['event']['properties']['part'].get('messageID') == completed[0][1]['id']
                       and row['event']['properties']['part'].get('sessionID') == session
                       and row['event']['properties']['part'].get('type') == 'tool'
                       and row['event']['properties']['part'].get('tool') == 'read'
                       and row['event']['properties']['part'].get('state', {}).get('status') == 'completed'
                       and row['event']['properties']['part']['state'].get('input', {}).get('filePath') == fixture_path
                       and 'Owned fixture.' in row['event']['properties']['part']['state'].get('output', '')
                       for row in events), 'fixture read did not complete'
        assert any(row.get('event', {}).get('type') == 'message.part.updated'
                   and row['event']['properties']['part'].get('messageID') == completed[-1][1]['id']
                   and row['event']['properties']['part'].get('sessionID') == session
                   and row['event']['properties']['part'].get('type') == 'text'
                   and row['event']['properties']['part'].get('text') == 'FIXTURE RESPONSE'
                   for row in events), 'missing correlated final text'
        outcomes.append(dict(user_message=ids[turn], session=session, chat_index=chat_index,
                             input_message_id=chat['input'].get('messageID'),
                             completed_stop_messages=[m['id'] for _, m in completed],
                             completed_indices=[i for i, _ in completed], idle_index=idle[0]))
    assert any(r['response'] == 'tool-stop' for r in requests), 'tool case never requested'
    assert any(r['response'] == 'after-tool' for r in requests), 'tool loop never continued'
    assert not any(row.get('event', {}).get('type') == 'session.error' for row in events)
    return outcomes


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--fux', type=Path, required=True)
    parser.add_argument('--opencode', type=Path, required=True)
    parser.add_argument('--output', type=Path, required=True)
    args = parser.parse_args()
    fux = args.fux.resolve(strict=True)
    opencode = args.opencode.resolve(strict=True)
    args.output.mkdir(mode=0o700, parents=True, exist_ok=False)
    requests, submissions, captures = [], [], []
    with tempfile.TemporaryDirectory(prefix='zoe-', dir='/tmp') as directory:
        root = Path(directory).resolve()
        for name in ['config/fux', 'cache', 'data', 'state', 'work']:
            (root / name).mkdir(parents=True)
        (root / 'work/fixture.txt').write_text('Owned fixture.\n')
        plugin = root / 'probe.js'
        plugin.write_text(PLUGIN)

        class Provider(BaseHTTPRequestHandler):
            def log_message(self, *_):
                pass

            def do_POST(self):
                self.connection.settimeout(3)
                size = int(self.headers.get('Content-Length', '0'))
                if self.path != '/v1/chat/completions' or not 0 < size <= 1048576 or len(requests) >= 8:
                    self.send_error(400)
                    return
                body = json.loads(self.rfile.read(size))
                messages = body['messages']
                tool_turn = any('ZOR_TOOL_STOP' in str(m.get('content')) for m in messages if m['role'] == 'user')
                tool_result = any(m['role'] == 'tool' for m in messages)
                label = 'after-tool' if tool_result else 'tool-stop' if tool_turn else 'text'
                delta = {'role': 'assistant', 'content': 'FIXTURE RESPONSE'}
                if label == 'tool-stop':
                    delta = {'role': 'assistant', 'tool_calls': [dict(index=0, id='fixture-read', type='function',
                        function=dict(name='read', arguments=json.dumps({'filePath': str(root / 'work/fixture.txt')})))]}
                requests.append(dict(response=label, roles=[m['role'] for m in messages],
                                     model=body['model'], stream=body.get('stream')))
                base = dict(id='chatcmpl-fixture', object='chat.completion.chunk', created=1, model='fixture')
                chunks = [dict(base, choices=[dict(index=0, delta=delta, finish_reason=None)]),
                          dict(base, choices=[dict(index=0, delta={}, finish_reason='stop')])]
                payload = (''.join('data: ' + json.dumps(c) + '\n\n' for c in chunks) + 'data: [DONE]\n\n').encode()
                self.send_response(200)
                self.send_header('Content-Type', 'text/event-stream')
                self.send_header('Content-Length', str(len(payload)))
                self.end_headers()
                self.wfile.write(payload)

        class BoundedHTTPServer(HTTPServer):
            def get_request(self):
                connection, address = super().get_request()
                # Bound header parsing too, before BaseHTTPRequestHandler runs.
                connection.settimeout(3)
                return connection, address

        provider = BoundedHTTPServer(('127.0.0.1', 0), Provider)
        thread = threading.Thread(target=provider.serve_forever)
        thread.start()
        process = None
        pane = instance = None
        control = root / 'fux/default.sock'
        evidence = dict(schema=1, captured_at=datetime.now(timezone.utc).isoformat(),
                        platform=os.uname().sysname, prompts=PROMPTS,
                        limitation='Real OpenCode TUI with synthetic local provider. No zor report binding, cloud credentials, model-quality or task-success claim.',
                        requests=requests, submissions=submissions, captures=captures)

        def value(command, **fields):
            return rpc(control, dict(id=1, command=command, instance=instance, **fields))

        def until(function):
            deadline = time.monotonic() + 30
            while time.monotonic() < deadline:
                result = function()
                if result:
                    return result
                if process.poll() is not None:
                    raise RuntimeError('fux exited before observation')
                time.sleep(.1)
            raise TimeoutError('fixture observation exceeded 30 seconds')

        try:
            config = dict(plugin=[plugin.as_uri()], model='fixture/fixture', small_model='fixture/fixture',
                          enabled_providers=['fixture'], permission={'*': 'deny', 'read': 'allow'}, provider={'fixture': dict(
                npm='@ai-sdk/openai-compatible', name='Local Fixture',
                options=dict(baseURL=f'http://127.0.0.1:{provider.server_port}/v1', apiKey='fixture'),
                models={'fixture': dict(name='Fixture', limit=dict(context=65536, output=1024))})})
            config_path = root / 'work/opencode.json'
            config_path.write_text(json.dumps(config))
            env = dict(PATH='/usr/bin:/bin:/usr/sbin:/sbin', HOME=str(root), TMPDIR=str(root),
                       XDG_RUNTIME_DIR=str(root), XDG_CONFIG_HOME=str(root / 'config'),
                       XDG_CACHE_HOME=str(root / 'cache'), XDG_DATA_HOME=str(root / 'data'),
                       XDG_STATE_HOME=str(root / 'state'), TERM='xterm-256color', SHELL='/bin/sh', LANG='en_US.UTF-8',
                       OPENCODE_CONFIG=str(config_path), ZOR_PROBE_LOG=str(root / 'events.jsonl'),
                       OPENCODE_DISABLE_DEFAULT_PLUGINS='true', OPENCODE_DISABLE_MODELS_FETCH='true',
                       OPENCODE_DISABLE_AUTOUPDATE='true')
            version = subprocess.run([str(opencode), '--version'], env=env, cwd=root / 'work',
                                     capture_output=True, timeout=10, check=True).stdout.decode().strip()
            evidence.update(opencode_version=version, opencode_sha256=sha256(opencode),
                            fux_sha256=sha256(fux), harness_sha256=sha256(Path(__file__)),
                            plugin_sha256=hashlib.sha256(PLUGIN.encode()).hexdigest())
            (root / 'config/fux/config.toml').write_text('default-command = { argv = ' +
                json.dumps([str(opencode), str(root / 'work'), '--model', 'fixture/fixture']) + ' }\n')
            with (root / 'stderr').open('wb') as err:
                process = subprocess.Popen([str(fux), 'serve'], env=env, cwd=root / 'work',
                                           stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL, stderr=err)
            until(control.exists)
            listing = rpc(control, dict(id=1, command='list'))
            instance = listing['instance']
            pane = listing['workspaces'][0]['tabs'][0]['panes'][0]['id']

            def capture():
                return value('capture', pane=pane, attrs=False, scrollback=0, max_bytes=131072)

            def ready():
                screen = capture()
                return screen if 'Fixture' in screen['text'] and 'Ask anything' in screen['text'] else None

            captures.append(until(ready))
            for turn, prompt in enumerate(PROMPTS):
                reservation = value('input-reserve', pane=pane)['receipt']
                submissions.append(value('input-submit', operation=reservation['operation'], keys=prompt + '\r'))

                def finished():
                    events = read_events(root / 'events.jsonl')
                    return events if sum(row.get('event', {}).get('type') == 'session.idle' for row in events) >= turn + 1 else None

                until(finished)
                submissions[-1]['final_receipt'] = value('input-status', operation=reservation['operation'])['receipt']
                captures.append(capture())
        finally:
            try:
                if process is not None:
                    try:
                        if pane is not None and process.poll() is None:
                            value('kill', pane=pane)
                    finally:
                        if process.poll() is None:
                            process.terminate()
                        try:
                            process.wait(timeout=10)
                        except subprocess.TimeoutExpired:
                            process.kill()
                            process.wait(timeout=5)
                            evidence['cleanup'] = 'forced kill; child cleanup unconfirmed'
                            raise
                        evidence['fux_exit'] = process.returncode
            finally:
                provider.shutdown()
                provider.server_close()
                thread.join(timeout=5)
                evidence['provider_thread_stopped'] = not thread.is_alive()
                evidence['events'] = read_events(root / 'events.jsonl')
                encoded = json.dumps(evidence, indent=2, ensure_ascii=False).replace(str(root), '<FIXTURE_ROOT>')
                (args.output / 'diagnostic.json').write_text(encoded + '\n')
        assert evidence['fux_exit'] == 0 and evidence['provider_thread_stopped'], 'cleanup failed'
        evidence['outcomes'] = validate(evidence['events'], submissions, requests, str(root / 'work/fixture.txt'))
        encoded = json.dumps(evidence, indent=2, ensure_ascii=False).replace(str(root), '<FIXTURE_ROOT>')
        (args.output / 'events.json').write_text(encoded + '\n')
        print(args.output / 'events.json')


if __name__ == '__main__':
    main()
