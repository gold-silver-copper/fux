    #[test]
    #[ignore = "manual bounded headless cost measurement; no acceptance timing threshold"]
    fn measure_native_milestone_view_costs() -> Result<(), Box<dyn std::error::Error>> {
        use std::{hint::black_box, time::Instant};
        const ITERATIONS: u32 = 200;
        for (name, line) in [("ascii", "x".repeat(80)), ("unicode", "界".repeat(40))] {
            let mut parser = vt100::Parser::new(24, 80, 0);
            for row in 1..=24 {
                parser.process(format!("\x1b[{row};1H{line}").as_bytes());
            }
            let view = PaneView::from_screen(parser.screen(), "cost probe", 0, None)?;
            let encoded_bytes = serde_json::to_vec(&view)?.len();
            for repetition in 1..=3 {
                let start = Instant::now();
                for _ in 0..ITERATIONS {
                    black_box(black_box(&view));
                }
                let loop_ns = start.elapsed().as_nanos();
                let start = Instant::now();
                for _ in 0..ITERATIONS {
                    black_box(PaneView::from_screen(
                        black_box(parser.screen()),
                        "cost probe",
                        0,
                        None,
                    )?);
                }
                let construction_ns = start.elapsed().as_nanos();
                let start = Instant::now();
                for _ in 0..ITERATIONS {
                    assert!(black_box(&view).valid());
                }
                let validation_ns = start.elapsed().as_nanos();
                let start = Instant::now();
                for _ in 0..ITERATIONS {
                    black_box(serde_json::to_vec(black_box(&view))?);
                }
                let serialization_ns = start.elapsed().as_nanos();
                println!(
                    "{}",
                    serde_json::json!({"workload":name,"repetition":repetition,
                    "iterations":ITERATIONS,"geometry":[80,24],"encoded_bytes":encoded_bytes,
                    "loop_ns":loop_ns,"construction_ns":construction_ns,
                    "validation_ns":validation_ns,"serialization_ns":serialization_ns})
                );
            }
        }
        Ok(())
    }
